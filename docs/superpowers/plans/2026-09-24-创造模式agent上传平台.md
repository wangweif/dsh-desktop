# 创造模式 agent 上传到 agent_platform 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** dsh-desktop 平台智能体页新增「我的创建」区，把创造模式产出的本地 preset 一键上传到 agent_platform（上传即发布，落本租户私有场景，全租户可下载），tenant_admin 可事后改名/删除。

**Architecture:** 平台侧新增 `POST /api/agents/upload`（DB 写路径，与 AI 探索发布同构；无 source_slug → 读路径走回退档，列表/export/自动同步零改动），复用 `_ensure_tenant_scene` 落私有场景。桌面侧 `EnterpriseAgentStore` 扩 `listLocalPresets`/`uploadAgent`（persona 提取 + `uploaded-agent.json` 映射），`auth.apiPost` 出 POST 能力，`agents.html` 加上传 UI。

**Tech Stack:** agent_platform（FastAPI + SQLAlchemy + Jinja2/Alpine，无测试基建、curl 验收）；dsh-desktop（Electron + TypeScript + vitest，`yaml` 包 eemeli/yaml v2）。

**设计文档:** `docs/superpowers/specs/2026-09-24-创造模式agent上传平台-design.md`

# --- 智能体上传（DSH Desktop 创造模式产物 → 本租户私有场景；上传即发布） ---


@router.post("/agents/upload")
async def upload_agent(request: Request, user: User = Depends(get_current_user)):
    """桌面端「创造模式」本地 preset 上传：写 DB 行（无 source_slug，读路径走回退档），
    落本租户私有场景 → 全租户立即可见可下载；tenant_admin 经 /tenant/agents 事后改名/删除。

    请求体 {name, description, system_prompt, agent_id?}；agent_id 有值 = 更新（仅上传者
    本人，version+1，updated_at 由 onupdate 自动刷新）。响应契约对齐桌面端 agents.ts。
    """
    body = await request.json()
    name = str(body.get("name") or "").strip()
    description = str(body.get("description") or "").strip()
    system_prompt = str(body.get("system_prompt") or "").strip()
    if not name or len(name) > 64:
        return _error("智能体名称不能为空且不超过 64 字", 400)
    if len(description) > 2000:
        return _error("描述不能超过 2000 字", 400)
    if not system_prompt or len(system_prompt) > 100_000:
        return _error("系统提示词不能为空且不能超过 100000 字", 400)
    if not user.tenant_id:
        return _error("上传智能体需要属于一个租户的账号", 403)

    agent_id = body.get("agent_id")
    async with async_session() as db:
        agent = None
        if agent_id:
            try:
                aid = uuid.UUID(str(agent_id))
            except (ValueError, TypeError):
                return _error("智能体 ID 无效", 400)
            # 必须在本租户私有场景内（防越权更新平台级智能体）
            agent = await _check_tenant_agent(db, aid, user.tenant_id)
            if agent is None:
                return _error("智能体不存在", 404)
            if str(agent.user_id) != str(user.id):
                return _error("只有上传者本人可以更新该智能体", 403)
        if agent is None:
            agent = AgentDefinition(
                user_id=user.id,
                name=name,
                description=description,
                system_prompt=system_prompt,
                agent_type="chat",
                is_builtin=False,
            )
            db.add(agent)
            await db.commit()
            await db.refresh(agent)
            scene = await _ensure_tenant_scene(db, user.tenant_id)
            db.add(ScenarioAgent(scenario_id=scene.id, agent_id=agent.id))
        else:
            agent.name = name
            agent.description = description
            agent.system_prompt = system_prompt
            agent.version = (agent.version or 1) + 1
        await db.commit()
        await db.refresh(agent)
        return _success({
            "id": str(agent.id), "name": agent.name, "version": agent.version,
            "updated_at": agent.updated_at.isoformat() if agent.updated_at else None,
        })

`src/shared/persona-extract.ts`：

```ts
import { isMap, isSeq, parseDocument, type Document, type Node, type YAMLMap } from 'yaml'
import {
  JS_TAG,
  resolveNode,
  rowName,
  scalarString,
  staticallyDisabled
} from './persona-prefix'

/** 从本地 agent.cordis.yml 提取 persona 系统提示词（上传平台用）。 */
export interface ExtractedPersona {
  /** persona 行 config.prefix 的字符串值（trim 后）；不可得为 null */
  prompt: string | null
  /** prompt 为 null 时的原因：用户提示与测试断言共用 */
  reason: 'ok' | 'parse-error' | 'no-persona' | 'empty-prompt'
}

const PERSONA_PACKAGE = '@deepseek-ai/dsh-persona'
const BOM = '﻿'

/**
 * 取第一个未被禁用、name 为 dsh-persona 的行的 config.prefix 字符串。
 * !!js 行不求值原样容忍；group 嵌套同 walk。只认 `prefix`——旧 `text` 键
 * 由 repair 流程迁移，此处不兜底（migratePersonaPrefix 的职责）。
 */
export function extractPersonaPrompt(source: string): ExtractedPersona {
  const body = source.startsWith(BOM) ? source.slice(1) : source
  const document: Document = parseDocument(body, { customTags: [JS_TAG] })
  if (document.errors.length > 0) return { prompt: null, reason: 'parse-error' }
  const stack = new Set<Node>()
  let found: string | null = null
  const walk = (node: Node | null, ancestorDisabled: boolean): void => {
    const seq = resolveNode(node, document, stack)
    if (!isSeq(seq) || stack.has(seq)) return
    stack.add(seq)
    for (const item of seq.items) {
      const row = resolveNode(item, document, stack)
      if (!isMap(row) || stack.has(row)) continue
      stack.add(row)
      const rowDisabled = ancestorDisabled || staticallyDisabled(row.get('disabled', true), document, stack)
      const config = resolveNode(row.get('config', true), document, stack)
      if (
        found === null &&
        !rowDisabled &&
        rowName(row as YAMLMap, document, stack) === PERSONA_PACKAGE &&
        isMap(config)
      ) {
        const prefix = scalarString(resolveNode((config as YAMLMap).get('prefix', true), document, stack))
        if (prefix !== undefined) found = prefix
      }
      if (isSeq(config)) walk(config, rowDisabled)
      stack.delete(row)
    }
    stack.delete(seq)
  }
  walk(document.contents, false)
  if (found === null) return { prompt: null, reason: 'no-persona' }
  const prompt = found.trim()
  if (prompt.length === 0) return { prompt: null, reason: 'empty-prompt' }
  return { prompt, reason: 'ok' }
}
```

  /** 与 apiGet 同一套包络/鉴权/超时语义的 POST 版；cookie 同样不出本类。 */
  async apiPost(path: string, body: unknown): Promise<EnterpriseApiResult> {
    const cookie = this.#sessionCookie
    if (!cookie) return { status: 'unauthorized' }
    let response: Response
    try {
      response = await this.#fetchImpl(`${this.#serverUrl}${path}`, {
        method: 'POST',
        headers: { cookie, 'content-type': 'application/json' },
        body: JSON.stringify(body),
        signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS)
      })
    } catch (error) {
      this.#log(`[enterprise] api post ${path} failed: ${errorMessage(error)}`)
      return { status: 'unreachable' }
    }
    if (response.status === 401 || response.status === 403) return { status: 'unauthorized' }
    if (!response.ok) return { status: 'unreachable' }
    const payload = await parseJsonBody(response)
    if (payload?.success !== true) {
      return {
        status: 'error',
        code: typeof payload?.code === 'number' ? payload.code : response.status,
        message:
          typeof payload?.message === 'string' && payload.message.length > 0
            ? payload.message
            : `请求失败（${path}）`
      }
    }
    return { status: 'ok', data: payload.data }
  }

   * 上传本地 preset 到平台（上传即发布）：persona 提示词 + 元数据 → POST
   * /api/agents/upload；已有映射走更新（version+1），平台侧已被管理员删除
   * （404 包络）时回退为重新创建（新 id，version 从 1 起）。
   */
  async uploadAgent(auth: EnterpriseAuth, presetId: string): Promise<AgentUploadResult> {
    if (typeof presetId !== 'string' || !PRESET_ID.test(presetId) || presetId.startsWith(PLATFORM_PRESET_PREFIX)) {
      return { ok: false, code: 'invalid', message: '该智能体不支持上传' }
    }
    const dir = join(this.#presetRoot, presetId)
    let composition: string
    try {
      composition = await readFile(join(dir, COMPOSITION_FILE), 'utf8')
    } catch {
      return { ok: false, code: 'invalid', message: '未找到该智能体的组合文件' }
    }
    const persona = extractPersonaPrompt(composition)
    if (persona.prompt === null) {
      const why = persona.reason === 'parse-error' ? '组合文件解析失败' : '组合文件中没有可用的 persona 提示词'
      return { ok: false, code: 'invalid', message: `${why}，无法上传` }
    }
    const meta = await readPresetMetadata(dir, presetId)
    if (meta.name.trim().length === 0 || meta.name.length > 64) {
      return { ok: false, code: 'invalid', message: '智能体名称为空或超过 64 字，请先在 preset.yml 里修正' }
    }
    const link = await this.#readUploadedLink(dir)
    const payload: Record<string, unknown> = {
      name: meta.name,
      description: meta.description,
      system_prompt: persona.prompt
    }
    if (link) payload.agent_id = link.agentId
    let result = await auth.apiPost('/api/agents/upload', payload)
    if (result.status === 'error' && result.code === 404 && link) {
      delete payload.agent_id
      result = await auth.apiPost('/api/agents/upload', payload)
    }
    if (result.status !== 'ok') {
      return mapApiFailure(result, auth.getServerUrl(), '上传智能体失败')
    }
    const data =
      typeof result.data === 'object' && result.data !== null ? (result.data as Record<string, unknown>) : {}
    const agentId = text(data.id)
    if (!agentId) {
      return { ok: false, code: 'invalid', message: '平台返回的上传结果不完整' }
    }
    const manifest: UploadedManifest = {
      source: 'agent_platform_upload',

    }
